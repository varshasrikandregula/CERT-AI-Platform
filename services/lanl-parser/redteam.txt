from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI(title="LANL Parser")


class LANLRecord(BaseModel):
    timestamp: int
    user: str
    source_computer: str
    destination_computer: str


# Load red team events once at startup
redteam_events = set()

try:
    with open("redteam.txt", "r") as f:
        for line in f:
            line = line.strip()
            if line:
                redteam_events.add(line)
except FileNotFoundError:
    print("redteam.txt not found")


@app.get("/")
def home():
    return {"status": "LANL Parser Running"}


@app.post("/parse")
def parse_lanl(record: LANLRecord):

    event_string = (
        f"{record.timestamp},"
        f"{record.user},"
        f"{record.source_computer},"
        f"{record.destination_computer}"
    )

    if event_string in redteam_events:
        risk = "High"
        threat = "Red Team Activity Detected"
    else:
        risk = "Low"
        threat = "Normal Activity"

    return {
        "timestamp": record.timestamp,
        "user": record.user,
        "source_computer": record.source_computer,
        "destination_computer": record.destination_computer,
        "risk_level": risk,
        "threat_status": threat
    }
